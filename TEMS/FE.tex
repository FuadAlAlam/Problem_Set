\prob{}{\textbf{(Korea Summer Program Junior P2)}
Find all functions $f:\mathbb{R}\rightarrow\mathbb{R}$ such that$$f(y^2f(x))=xyf(y)$$for all $x,y\in\mathbb{R}$.}

\prob{}{\textbf{(Kosovo 2024 P3)}
Find all functions $f: \mathbb{R}\rightarrow \mathbb{R}$ such that$$f(x + yf(x) + f(y)) = xy + f(f(x))$$for all $x,y\in\mathbb{R}$.}

\prob{}{\textbf{(BdMO 2024 Higher Secondary P7)}
Find all functions $f:\mathbb{N} \to \mathbb{N}$ such that$$f\left(\Big \lceil \frac{f(m)}{n} \Big \rceil\right)=\Big \lceil \frac{m}{f(n)} \Big \rceil$$for all $m,n \in \mathbb{N}$.}

\prob{}{\textbf{(IFEO Shortlist 2020 A1)}
Find all functions $f: \mathbb{Z} \to \mathbb{Z}$ such that$$1+f^{f(a)+b}(a)=f(a)+f(a+b)$$for all $a,b\in\mathbb{R}$.}

\prob{}{\textbf{(USAJMO 2024 P5)}
Find all functions $f:\mathbb{R}\rightarrow\mathbb{R}$ such that$$f(x^2-y)+2yf(x)=f(f(x))+f(y)$$for all $x,y\in\mathbb{R}$.}

\prob{}{\textbf{(FEOO Shortlist A1)}
Find all functions $f:\mathbb{R}\rightarrow\mathbb{R}$ such that$$2xyf(xy)\geq (xy)^2+xf(x)f(y^2)$$for all $x,y\in\mathbb{R}$.}

\prob{}{\textbf{(IFEO 2020 P5)}
Find all functions $f:\mathbb{R}\rightarrow\mathbb{R}$ such that$$f(x^2+f(x)+y)=xf(x)+f(x+y)$$for all $x,y\in\mathbb{R}$.}

\prob{}{\textbf{(FEcreator handout)}
Find all functions $f:\mathbb{R}^+\rightarrow\mathbb{R}^+$ such that$$f(2x + 2f(y)) = x + f(x) + 2y$$for all $x, y \in \mathbb{R}^+$.}
