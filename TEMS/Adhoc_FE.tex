\prob{}{
    Let $f(x) = x^{67-x^{67-x^{67-\ldots}}}$, where $x \neq 0$. If $f(n) = 64$, then what is the remainder if $n^n$ is divisible by $11$?
}

\prob{}{
    Let$$f(x) + 2f(-x) = 2x$$and$$2f(x)+f(-x) = x^2$$How many $x$ are there between $1$ and $2024$ for which both $x$ and $f(x)$ are integers?
}

\prob{}{
    Let$$f(x+y) = f(x)+f(y)$$If $f(1) = 3$, find the value of $f(30)$.
}

\prob{}{
    Let $f(x, y) = y - x^2$, where $|x|+|y| \leq 21$. Find the difference between the largest and the smallest values of the function. ($x$ and $y$ both are integers)
}

\prob{}{
    If $f(n) = f(n+1)+f(n-1)$, $f(1) = 5$, $f(2) = 4$, then $f(2024) =$ ?
}

\prob{}{
    Let $f(1) = 2, f(2n) = f(2n-1) + 1$ and $f(2n-1) = f(2n-3)+2$ (when $n$ is natural number). Find $f(2023)$.
}

\prob{}{
    If $f(x+3) = f(x+1)+4x+12$ and $f(0) = 2$, then what is $f(500)$?
}

\prob{}{
    Let $S = \{1, 2, 3, \ldots, 12\}$. How many functions $f: S \rightarrow S$ are there such that $f(f(x)) = x$ and $f(x)-x$ is divisible by $3$?
}

\prob{}{
    If $f(x) = 3x-2$, then what is the value of x in the following equation where $f$ is $2020$ times iterated?$$x = f(f(f(\ldots f(x) \ldots)))$$
}

\prob{}{
    If $f(n) = f(n-1)+2n-2$ and $f(1) = 2$, then what is the value of $f(100)$?
}

\prob{}{
    Let$$f(3n) = 3f(n)$$ $$f(3n+1) = 3f(n)-1$$ $$f(3n+2) = 3f(n)-2$$
    If $f(1) = f(2) = 1, f(1009) = ?$
}

\prob{}{
    If $f(x) = 2019$, then what is the value of $f^{1019}(2019)$? \textbf{Note:} $f^2(x) = f(f(x))$.
}

\prob{}{
    Two functions $f(x, y)$ and $g(x)$ are defined on real numbers, such that$$f(x+y, x-y) = x \cdot g(y)+ y \cdot g(x)$$Given that, $g(x) = 5$ for all values of $x$. Find the value of $f(2019, 0)$.
}

\prob{}{
    Let $f(1) = 2, f(2) = 5, f(3) = 10$ and$$f(x) - f(x-3) = 2[f(x-1) - f(x-2)]$$ where $x>3$. $f(2019) = ?$
}

\prob{}{
    Let $f(4x-3) = 2x+5.$ If $f(2x+1) = ax+b$, $a+b = ?$
}

\prob{}{
    Suppose $f(x)$ is a function. It is given that$$f(2n-1) = f(2n-2)-1$$and$$f(2n) = 2f(2n-1)$$If $f(1) = n$, then what is the value of $n$ such that $f(1971)-f(2018) = -1?$
}

\prob{}{
    If $f(x^2-2016x) = xf(x)+2016$, what is the value of $f(2017)$?
}

\prob{}{
    Let$$f(x) = g(x+1)$$ $$f(y) = 2018y+2016y+\ldots+2y$$ $$g(x) = h(2x)-x$$If $h(2018) = a^3$ and $a$ is an integer, $3a = ?$
}

\prob{}{
    For any rational numbers $x, y$, function $f(x)$ is a real number and$$f(x+y) = f(x)f(y)-f(xy)+1$$Again, $f(2017) \neq f(2018)$, $f(\frac{2017}{2018}) = \frac{a}{b}$, where $a, b$ are co-prime. $a+b = ?$
}

\prob{}{
    If $F_0(x) = \frac{1}{1-x}$ and $F_n(x) = F_0(F_{n-1}(x))$. Find $F_{2015}(2016)$.
}

\prob{}{
    Let$$f(3m) = \frac{mf(m)}{3}$$ $$f(3m+2) = \frac{(m+2)f(m+2)}{3}$$ $$f(3m+1) = \frac{(m+1)f(m+1)}{3}$$ $f(2016) = ?$
}

\prob{}{
    For all positive integers $x$, $f(f(x)) = 4x+3$ and for one positive value of integer $k$, $f(5^k) = 5^k \times 2^{k-2} + 2^{k-3}$. $f(2015) = ?$
}

\prob{}{
    Let $f(y) = y$ repeated $y$ times, for example $f(3) = 333, f(5) = 55555$. If $a = f(2001)+f(2002)+\ldots+f(2014)+f(2015)$. What is the remainder upon division of $a$ by $3$?
}

\prob{}{
    $f(n) =$ sum of the squares of digits of $n$. If $f_k(n) = f(f_{k-1}(n))$, $f_{2014}(3) = ?$
}

\prob{}{
    For all positive integers $x, y$, $f(x) \geq 0$ and $f(xy) = f(x) + f(y)$. If the digit at the one's of $x$ is $6$, then $f(x) = 0$. If $f(1920) = 420$, then $f(2015) = ?$
}

\prob{}{
    If $x, a, b$ are positive integers, $f(x)$ is positive integer too, and if $a>b$, then $f(a)>f(b)$. Again, $f(f(x)) = x^2+2$. Find the value of $f(6)$.
}

\prob{}{
    A function $f: \mathbb{R} \rightarrow \mathbb{R}$ is defined in such a way that $f(x+y) = f(x) \cdot f(y)$, and for $a \in \mathbb{N}$, $\sum_{k=1}^n f(a+k) = 16(2^n-1)$. If $f(1) = 2$, then what is the value of $a$?
}

\prob{}{
    Given that, $F(x, y) = F(xy, \frac{x}{y})$ where $y \neq 0$. If $F(x^4, y^4)+F(x^2, y^2) = 16$, then $F(x, y) = ?$
}

\prob{}{

}

\prob{}{
    A function $f: \mathbb{R} \rightarrow \mathbb{R}$ satisfies $f(1) = 1$ and for any $x \in \mathbb{R}$, $f(x+7) \geq f(x)+7$ and $f(x+1) \leq f(x)+1$. $f(2013) = ?$
}

\prob{}{
    Let $f: \mathbb{R} - \{0\} \rightarrow \mathbb{R}$ be a function such that $f(xy) = \frac{f(x)}{y}$. If $f(2012) = 1$, then $f(2013) = ?$

}
