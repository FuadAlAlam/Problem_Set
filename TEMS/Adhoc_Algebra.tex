\prob{}{

}

\prob{}{
    Let $x$ and $y$ be real numbers such that $x \neq 1$, $y \neq 1$ and$$(x-1)^2 + (y-1)^2 = xy(x + y - 3) +1$$Show that $x + y = 1$.
}

\prob{}{
    Find all real quadruples $(a, b, c, d)$ satisfying\begin{align*}
        (b+c+d)^{2026}=3a \\
        (a+c+d)^{2026}=3b \\
        (a+b+d)^{2026}=3c \\
        (a+b+c)^{2026}=3d
    \end{align*}
}

\prob{}{
    Assign a positive integer to each face of a cube. At each vertex take the product of the labels on its three incident faces. These eight products sum to $1001$. Determine the sum of the six face labels.
}

\prob{}{
    Let $a, b, c, d$ and $e$ be positive integers such that$$abcde=a+b+c+d+e$$Find the maximum possible value of $\max \{a, b, c, d, e\}$.
}

\prob{}{

}

\prob{}{

        bcd+bc+cd+db+b+c+d=9 \\
        cda+cd+da+ac+c+d+a=9 \\
        dab+da+ab+bd+d+a+b=9
    \end{gather*}
}

\prob{}{

\prob{}{

}
