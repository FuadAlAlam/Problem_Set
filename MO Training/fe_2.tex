\prob{}{\textbf{(Centroamerican and Caribbean MO 2020 P3)}
Find all functions $f: \mathbb{Z}\to \mathbb{Z}$ such that$$f(a)+f(b)+f(c)=a^2+b^2+c^2$$ for all $a, b, c \in \mathbb{Z}$ satisfying $a+b+c = 0$.}

\prob{}{\textbf{(Baltic Way 2025 P18)}
Find all functions $f: \mathbb{Z}_{>0} \rightarrow \mathbb{Z}_{>0}$ such that $f(2)=1$ and$$\operatorname{lcm}(f(a+b), f(b)) \mid \operatorname{lcm}(a+ f(b), b)$$for all $a, b \in \mathbb{Z}_{>0}$.}

\prob{}{\textbf{(Pan American Girls' Mathematical Olympiad 2024 P5)}
Find all functions $f: \mathbb{R} \to \mathbb{R}$ such that$$f(f(x+y) - f(x)) + f(x)f(y) = f(x^2) - f(x+y)$$for all $x, y \in \mathbb{R}$.}

\prob{}{\textbf{(Rioplatense MO 2024 P5)}
Let $S = \{2, 3, 4, \dots\}$ be the set of positive integers greater than 1. Find all functions $f : S \to S$ such that$$\text{gcd}(a, f(b)) \cdot \text{lcm}(f(a), b) = f(ab)$$for all $a, b \in S$.}

\prob{}{\textbf{(Swiss MO 2020 P1)} Find all functions $f: \mathbb N\to \mathbb N$ such that$$f(m)+f(n)\mid m+n$$for all $m, n \in \mathbb N$.}

\prob{}{\textbf{(IMO 2022 P2)}
Find all functions $f: \mathbb{R}^+ \to \mathbb{R}^+$ such that for each $x \in \mathbb{R}^+$, there is exactly one $y \in \mathbb{R}^+$ such that$$xf(y)+yf(x) \leq 2$$}

\prob{}{\textbf{(Romanian Masters of Mathematics Shortlist 2021 A1)}
Find all functions $f:\mathbb{R} \to \mathbb{R}$ such that$$f(xy+f(x)) + f(y) = xf(y) + f(x+y)$$for all $x, y \in \mathbb{R}$.}

\prob{}{\textbf{(IMEO 2019 P3)}
Find all functions $f:\mathbb{R} \to \mathbb{R}$ such that$$(x+y) \cdot f(x+y)= f(f(x)+y) \cdot f(x+f(y))$$for all $x, y \in \mathbb{R}$.}
