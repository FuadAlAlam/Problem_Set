\prob{}{\textbf{(Rioplatense MO 2023 P1)}
Find all positive integers $(x, y, p)$ such that $p$ is prime, $p=x^2+1$ and $2p^2=y^2+1$.}

\prob{}{\textbf{(Pan American Girls' Mathematical Olympiad 2022 P2)}
Find all ordered triplets $(p, q, r)$ of positive integers such that $p$ and $q$ are two (not necessarily distinct) primes, $r$ is even, and$$p^3+q^2=4r^2+45r+103$$}

\prob{}{\textbf{(Kazakhstan Regional 2021 Grade 9 P3)}
Find all pairs $(x, y)$ of positive integers that satisfy the equation$$125 \cdot 2^x - 3^y = 271$$}

\prob{}{\textbf{(Belarus MO 2024 Grade 10 P6)}
Find all pairs $(p,q)$ of prime numbers satisfying the equation$$p^{2q+1} = q^{p} + 2023$$}

\prob{}{\textbf{(Al-Khwarizmi IJMO 2025 P1)}
Determine the largest integer $c$ for which there exists at least one triple $(x,y,z)$ of integers such that$$x^2 + 4(y + z) = y^2 + 4(z + x) = z^2 + 4(x + y) = c$$and all triples $(x,y,z)$ of real numbers satisfying the equations are such that $x,y,z$ are integers.}

\prob{}{\textbf{(USAJMO 2023 P1)}
Find all triples of positive integers $(x, y, z)$ that satisfy the equation$$2(x+y+z+2xyz)^2=(2xy+2yz+2zx+1)^2+2023$$}

\prob{}{\textbf{(Mongolia 2024 P1)}
Find all triples $(a, b, c)$ of positive integers such that $a \leq b$ and$$a!+b!=c^4+2024$$}

\prob{}{\textbf{(IMO 2022 P5)}
Find all triples $(a, b, p)$ of positive integers with $p$ prime and$$a^p=b!+p$$}
