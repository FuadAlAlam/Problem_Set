\prob{}{\textbf{(Turkey Junior 2025 P3)}
Find all positive integers $n$ such that the expression$$3n^3 - 7n^2 - 15n + 35$$is equal to the square of an integer.}

\prob{}{\textbf{(Korea 2023 P1)}
A sequence of positive reals $\{ a_n \}$ is defined below$$a_0 = 1, \quad a_1 = 3, \quad a_{n+2} = \frac{a_{n+1}^2+2}{a_n}$$Show that for all non-negative integers $n$, $a_n$ is a positive integer.}

\prob{}{\textbf{(Tuymaada 2022 Junior P5)}
Each row of a $24 \times 8$ table contains some permutation of the numbers $1, 2, \ldots , 8.$ In each column, the numbers are multiplied. What is the minimum possible sum of all the products?}

\prob{}{\textbf{(Benelux 2021 P3)}
A cyclic quadrilateral $ABXC$ has circumcentre $O$. Let $D$ be a point on line $BX$ such that $AD = BD$. Let $E$ be a point on line $CX$ such that $AE = CE$. Prove that the circumcenter of $\triangle DEX$ lies on the perpendicular bisector of $OA$.}

\prob{}{\textbf{(Belarus 2025 P5)}
Find the smallest positive integer $n$ such that both $n^3-n$ and $(n+1)^3-(n+1)$ are divisible by $2025$.}

\prob{}{\textbf{(Germany 2025 P1)}
Solve the system of equations in $\mathbb{R}$:
\begin{align*}
    \frac{a}{c} &= b-\sqrt{b}+c \\
    \sqrt{\frac{a}{c}} &= \sqrt{b}+1 \\
    \sqrt[4]{\frac{a}{c}} &=\sqrt[3]{b}-1
\end{align*}}

\prob{}{\textbf{(Indonesia 2025 P5)}
Let $A_1,A_2,\dots, A_n$ be subsets of $X = \{1,2,\dots, 22\}$, each with $15$ elements. It is known for each $B\subseteq X$ with $10$ elements, there exist $i,j \in \{1,2,\dots, n\}$ such that $B \subseteq A_i \cap A_j$. Prove that $n\ge 431$.}

\prob{}{\textbf{(Greece JBMO TST 2025 P2)}
Let $ABC$ be a right triangle with $\angle A =90^\circ$. Outside the triangle $ABC$, we construct an isosceles triangle $ACD$ such that $DA=DC=BC$. Lines $BC$ and $DA$ intersect at point $E$. Prove that $DC+DE=2 CE$.}
